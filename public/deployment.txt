Jolitos puslapio deploymentas: 

bash: 
# make sure you’re on main
git checkout main

# build the site
npm run build

# switch to deploy branch
git checkout deploy

# remove old deploy files
git rm -rf .

# copy fresh build files
cp -r dist/* .

# commit changes
git add .
git commit -m "Deploy $(date +'%Y-%m-%d %H:%M:%S')"

# push to GitHub
git push origin deploy

# switch back to main for development
git checkout main

#restore
 npm i --save-dev @types/react
